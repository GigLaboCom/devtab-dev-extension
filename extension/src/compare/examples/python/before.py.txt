"""Summaries of daily temperature readings."""
import csv
from statistics import mean


def load(path):
    with open(path, newline="") as f:
        return [row for row in csv.DictReader(f)]


def by_city(rows):
    cities = {}
    for row in rows:
        cities.setdefault(row["city"], []).append(float(row["celsius"]))
    return cities


def to_fahrenheit(celsius):
    return celsius * 9 / 5 + 32


def warmest(cities):
    return max(cities, key=lambda name: mean(cities[name]))


def report(cities):
    for name, values in sorted(cities.items()):
        print(f"{name}: avg {mean(values):.1f} C over {len(values)} days")


if __name__ == "__main__":
    rows = load("readings.csv")
    cities = by_city(rows)
    report(cities)
    print("warmest:", warmest(cities))
