# Services of the local development setup.
version: 2

services:
  web:
    image: local/web:1.8
    ports:
      - "3000:3000"
    environment:
      LOG_LEVEL: debug
      API_BASE: /api
    depends_on:
      - api

  api:
    image: local/api:2.1
    ports:
      - "4000:4000"
    environment:
      DATABASE_HOST: db
      DATABASE_POOL: 5
    depends_on:
      - db

  worker:
    image: local/api:2.1
    command: ["run", "jobs"]

  db:
    image: local/postgres:16
    volumes:
      - db-data:/var/lib/postgresql/data

volumes:
  db-data: {}
