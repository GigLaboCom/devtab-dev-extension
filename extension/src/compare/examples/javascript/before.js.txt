// Helpers for prices in a shopping cart.
const TAX_RATE = 0.2;

export function subtotal(lines) {
  return lines.reduce((sum, line) => sum + line.price * line.quantity, 0);
}

export function discount(amount, code) {
  if (code === 'WELCOME') return amount * 0.1;
  if (code === 'BULK' && amount > 500) return 50;
  return 0;
}

export function tax(amount) {
  return amount * TAX_RATE;
}

export function format(amount) {
  return amount.toFixed(2) + ' EUR';
}

export function total(lines, code) {
  const sum = subtotal(lines);
  const net = sum - discount(sum, code);
  return format(net + tax(net));
}

export function isEmpty(lines) {
  return lines.length === 0;
}
