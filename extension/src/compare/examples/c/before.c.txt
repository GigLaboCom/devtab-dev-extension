#include <stdbool.h>
#include <stddef.h>

#define CAPACITY 64

/* A fixed-size ring buffer of bytes. */
typedef struct {
    unsigned char data[CAPACITY];
    size_t head;
    size_t tail;
    size_t count;
} ring_t;

void ring_init(ring_t *r) {
    r->head = 0;
    r->tail = 0;
    r->count = 0;
}

bool ring_push(ring_t *r, unsigned char byte) {
    if (r->count == CAPACITY) {
        return false;
    }
    r->data[r->head] = byte;
    r->head = (r->head + 1) % CAPACITY;
    r->count++;
    return true;
}

bool ring_pop(ring_t *r, unsigned char *out) {
    if (r->count == 0) {
        return false;
    }
    *out = r->data[r->tail];
    r->tail = (r->tail + 1) % CAPACITY;
    r->count--;
    return true;
}

bool ring_empty(const ring_t *r) {
    return r->count == 0;
}
