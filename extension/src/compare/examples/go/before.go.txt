package queue

import (
	"errors"
	"sync"
)

// ErrEmpty is returned when there is nothing to take.
var ErrEmpty = errors.New("queue is empty")

// Queue is a first-in first-out list of jobs.
type Queue struct {
	mu   sync.Mutex
	jobs []string
}

// Push adds a job at the end.
func (q *Queue) Push(job string) {
	q.mu.Lock()
	defer q.mu.Unlock()
	q.jobs = append(q.jobs, job)
}

// Pop takes the first job.
func (q *Queue) Pop() (string, error) {
	q.mu.Lock()
	defer q.mu.Unlock()
	if len(q.jobs) == 0 {
		return "", ErrEmpty
	}
	job := q.jobs[0]
	q.jobs = q.jobs[1:]
	return job, nil
}

// Len reports how many jobs wait.
func (q *Queue) Len() int {
	q.mu.Lock()
	defer q.mu.Unlock()
	return len(q.jobs)
}
