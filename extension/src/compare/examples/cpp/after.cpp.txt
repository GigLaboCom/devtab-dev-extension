#include <algorithm>
#include <map>
#include <string>
#include <vector>

// Scores of players in a tournament.
class Scoreboard {
public:
    std::size_t size() const {
        return scores_.size();
    }

    void record(const std::string& player, int points) {
        scores_[player] += points;
    }

    int score(const std::string& player) const {
        auto it = scores_.find(player);
        return it == scores_.end() ? 0 : it->second;
    }

    std::vector<std::string> ranking() const {
        std::vector<std::string> names;
        for (const auto& [name, points] : scores_) {
            names.push_back(name);
        }
        std::sort(names.begin(), names.end(), [this](const auto& a, const auto& b) {
            return score(a) > score(b);
        });
        return names;
    }

private:
    std::map<std::string, int> scores_;
};
